# AWS Secrets Management

I help you manage secrets securely.

## SSM Parameter Store
```typescript
import { SSMClient, GetParameterCommand, PutParameterCommand } from '@aws-sdk/client-ssm';

const ssm = new SSMClient({ region: process.env.AWS_REGION });

// Get secret
async function getSecret(name: string): Promise<string> {
  const response = await ssm.send(new GetParameterCommand({
    Name: name,
    WithDecryption: true,
  }));
  return response.Parameter!.Value!;
}

// Store secret
async function putSecret(name: string, value: string) {
  await ssm.send(new PutParameterCommand({
    Name: name,
    Value: value,
    Type: 'SecureString',
    Overwrite: true,
  }));
}
```

## Secrets Manager
```typescript
import { SecretsManagerClient, GetSecretValueCommand } from '@aws-sdk/client-secrets-manager';

const sm = new SecretsManagerClient({ region: process.env.AWS_REGION });

async function getSecretJSON(secretId: string) {
  const response = await sm.send(new GetSecretValueCommand({
    SecretId: secretId,
  }));
  return JSON.parse(response.SecretString!);
}

// Usage
const dbCreds = await getSecretJSON('prod/db/credentials');
const connectionString = `postgres://${dbCreds.username}:${dbCreds.password}@...`;
```

## When to Use Which
| Service | Use Case |
|---------|----------|
| Parameter Store | Config values, feature flags, simple secrets |
| Secrets Manager | Database credentials, API keys with rotation |
