# Build ETL Pipeline

You are building a data pipeline.

## Python ETL
```python
import pandas as pd
from datetime import datetime
import logging

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

def extract(source: str) -> pd.DataFrame:
    """Extract data from source."""
    logger.info(f"Extracting from {source}")
    df = pd.read_csv(source)
    logger.info(f"Extracted {len(df)} rows")
    return df

def transform(df: pd.DataFrame) -> pd.DataFrame:
    """Clean and transform data."""
    logger.info("Transforming data")

    # Remove nulls
    df = df.dropna(subset=['required_column'])

    # Convert types
    df['date'] = pd.to_datetime(df['date'])
    df['amount'] = df['amount'].astype(float)

    # Add derived columns
    df['year_month'] = df['date'].dt.to_period('M')

    # Validate
    assert df['amount'].min() >= 0, "Negative amounts found"

    logger.info(f"Transformed to {len(df)} rows")
    return df

def load(df: pd.DataFrame, destination: str) -> None:
    """Load data to destination."""
    logger.info(f"Loading to {destination}")
    df.to_parquet(destination, index=False)
    logger.info("Load complete")

def run_pipeline():
    """Run the full ETL pipeline."""
    start_time = datetime.now()
    try:
        raw = extract('data/raw.csv')
        clean = transform(raw)
        load(clean, 'data/processed.parquet')
        logger.info(f"Pipeline completed in {datetime.now() - start_time}")
    except Exception as e:
        logger.error(f"Pipeline failed: {e}")
        raise
```
